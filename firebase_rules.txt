rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
  match /applications/{id} { allow create: if request.resource.data.status == "Under Review"; allow read,update,delete: if request.auth != null; }
  match /phoneLocks/{phone} { allow read: if true; allow create: if !exists(/databases/$(database)/documents/phoneLocks/$(phone)); allow update,delete: if false; }
  match /statusLookup/{id} { allow read: if true; allow create: if request.resource.data.status == "Under Review"; allow update,delete: if request.auth != null; }
  match /squad/{id} { allow read: if true; allow write: if request.auth != null; }
  match /settings/{id} { allow read: if true; allow write: if request.auth != null; }
 }
}
